  1. أضف params: وتحته region: "Jazan"، واكتب REGION <- params$region في مقطع الإعداد.
  2. ولِّد التقرير بالأمر quarto render report.qmd -P region:Riyadh واقرأ كل جملة: هل بقي شيء يذكر جازان؟
  3. اكتب render-all.R بالحلقة التي في المثال، لثلاث مناطق أولًا.
library(tidyverse)

regions <- read_csv("regions.csv") |> pull(region_en)

walk(regions, \(r) {
  slug <- r |> str_to_lower() |> str_replace_all(" ", "-")
  quarto::quarto_render(
    "report.qmd",
    output_format = "typst",
    execute_params = list(region = r),
    output_file = str_glue("report-{slug}.pdf")
  )
})
