```{r}
library(tidyverse)
rates <- read_csv("unemployment.csv") |>
filter(region != "Kingdom")
up <- rates |>
group_by(region) |>
summarise(change = last(rate) - first(rate)) |>
filter(change > 0) |>
pull(region)
jazan_end <- rates |>up_end <- rates |>
filter(region == "Jazan") |> filter(region %in% up) |>
slice_tail(n = 1)
```
