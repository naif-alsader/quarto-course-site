```{r}

library(tidyverse)
library(tinytable)
rates <- read_csv("unemployment.csv")
area <- rates |> filter(region == params$region)
kingdom <- rates |> filter(region == "Kingdom")
now <- last(area$rate)
change <- now - first(area$rate)
gap <- now - last(kingdom$rate)
```
