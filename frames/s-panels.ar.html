library(tidyverse)
rates <- read_csv("unemployment.csv") |>all <- read_csv("unemployment.csv")
rates <- all |>
filter(region != "Kingdom") filter(region != "Kingdom") |>
group_by(region) |>
mutate(change = last(rate) - first(rate)) |>
ungroup() |>
mutate(region = fct_reorder(region, change,
.desc = TRUE))
riser <- rates |>
filter(change > 0) |>
distinct(region) |>
pull(region)
kingdom <- all |>
filter(region == "Kingdom") |>
select(-region)
